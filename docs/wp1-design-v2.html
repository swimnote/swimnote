DO $$ BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint
    WHERE conname = 'fk_ge_curriculum_item'
      AND conrelid = 'growth_events'::regclass
  ) THEN
    ALTER TABLE growth_events
      ADD CONSTRAINT fk_ge_curriculum_item
      FOREIGN KEY (curriculum_item_id)
      REFERENCES curriculum_items(id)
      ON DELETE RESTRICT;
  END IF;
END $$;

DO $$ BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint
    WHERE conname = 'fk_ge_curriculum_version'

      ADD CONSTRAINT fk_ge_curriculum_version

§8 — Migration Group 순서 (정정)
§8-A · initXModeSchema() — PART 1 전용 정정 설계 
export async function initXModeSchema(db: Db): Promise {
  // Group 1: ENUM 3개 + swimming_pools xmode 컬럼 5개
  //   이미 DB 적용 → 멱등 재실행 (변경 없음)
  await runGroup1_EnumAndPools(db);

  // Group 2: global_template_sets + diary_templates 확장
  //   이미 DB 적용 → 멱등 재실행 (변경 없음)
  await runGroup2_GlobalTemplate(db);

  // Group 3: audit_entity_versions + next_audit_version() + audit_logs
  //   이미 DB 적용 → 멱등 재실행 (변경 없음)
  await runGroup3_Audit(db);

  // Group 5a: growth_events (M-I만)
  //   이미 DB 적용 → 멱등 재실행 (변경 없음)
  await runGroup5a_GrowthEvents(db);

§10-A · 멱등성 
-- 멱등 패턴 (전체 Group 공통)
ENUM       : DO $$ BEGIN CREATE TYPE ... EXCEPTION WHEN duplicate_object THEN NULL; END $$
테이블     : CREATE TABLE IF NOT EXISTS
컬럼       : ALTER TABLE ... ADD COLUMN IF NOT EXISTS
인덱스     : CREATE INDEX IF NOT EXISTS / CREATE UNIQUE INDEX IF NOT EXISTS
CONSTRAINT : DO $$ + pg_constraint 조회 후 없을 때만 ADD CONSTRAINT
함수       : CREATE OR REPLACE FUNCTION

-- 재실행 안전성
운영 DB (Group 1~5a 이미 적용): 전체 IF NOT EXISTS 통과 → 변경 없음
