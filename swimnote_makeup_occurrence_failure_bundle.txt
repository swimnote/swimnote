    // STEP 5: 동일 반+보강 조합 요청 진행 중 재호출 방지
    const occKey = `${activeTarget.id}:${classId}`;

    const mySeq = occSeqRef.current;

    setOccurrences([]);

    const _doRequest = async () => {
      const r = await apiRequest(token, `/teacher/makeups/${activeTarget.id}/eligible-occurrences?class_group_id=${classId}`);
      return r;
    };
    try {
      let r = await _doRequest();
      // 실패 시 1회 자동 재시도 (500ms 후)
      if (!r.ok) {

        r = await _doRequest();
      }
      if (occSeqRef.current !== mySeq) return;
      if (r.ok) {
        const data = await r.json();
        setOccurrences((data.occurrences || []) as MakeupOccurrence[]);
      } else {
        const status = r.status;

        let body: any = null;
        try { body = rawText ? JSON.parse(rawText) : null; } catch { body = { raw: rawText }; }
        const errorCode: string = body?.error ?? "UNKNOWN";

/** schedule_days("월,수,금") → 요일 번호 Set */
function parseDayNums(scheduleDays: string): Set<number> {
  const map: Record<string, number> = { 일: 0, 월: 1, 화: 2, 수: 3, 목: 4, 금: 5, 토: 6 };
  const nums = new Set<number>();
  scheduleDays.split(",").map(d => d.trim()).forEach(d => { if (map[d] !== undefined) nums.add(map[d]); });
  return nums;
}

export async function apiRequest(token: string | null, path: string, options: RequestInit = {}) {
  const url = `${_API_BASE}${path}`;
  const method = (options.method ?? "GET").toUpperCase();
  const isWrite = method !== "GET";

  // 쓰기 요청: 관련 캐시 즉시 삭제
  if (isWrite) _bustRelated(path);

  // GET: 캐시 히트 시 즉시 반환
  if (!isWrite) {
    try {
      const cacheKey = _makeCacheKey(token, path);
      const cached = _getCached(cacheKey);
      if (cached !== null) {
        const fakeRes = {
          ok: true, status: 200,
          json: async () => cached,
          clone: () => ({ ok: true, status: 200, json: async () => cached }),
        } as unknown as Response;
        return fakeRes;
      }
    } catch {}
  }

  const controller = new AbortController();
  const timeoutId = setTimeout(() => controller.abort(), 30_000);  // 30초 타임아웃
  let res: Response;
  try {
    res = await fetch(url, {
      cache: "no-store",
      ...options,
      signal: controller.signal,
      headers: {
        "Content-Type": "application/json",
        "Cache-Control": "no-cache",
        ...(token ? { Authorization: `Bearer ${token}` } : {}),
        ...options.headers,
      },
    });
  } catch (e: any) {
    clearTimeout(timeoutId);
    if (e?.name === "AbortError") throw new Error("요청 시간이 초과됐습니다. 네트워크 연결을 확인해주세요.");
    throw e;
  }
  clearTimeout(timeoutId);

  // GET 2xx 응답만 캐시 저장 (clone으로 원본 보존)
  if (!isWrite && res.ok) {
    try {
      const cacheKey = _makeCacheKey(token, path);
      const clonedForCache = res.clone();
      clonedForCache.json().then(data => _setCached(cacheKey, data)).catch(() => {});
    } catch {}
  }

  // 401: 탈퇴/삭제 계정 → 전역 강제 로그아웃
  if (res.status === 401) {
    try {
      const cloned = res.clone();
      const body = await cloned.json().catch(() => ({}));
      if (body?.error === "account_withdrawn" || body?.error === "account_deleted") {
        _globalLogoutHandler?.();
      }
    } catch {}
  }

  // 403: ROLE_REVOKED → teacher 모드 강제 복귀
  if (res.status === 403) {
    try {
      const cloned = res.clone();
      const body = await cloned.json().catch(() => ({}));
      if (body?.code === AuthErrorCodes.ROLE_REVOKED) {
        _globalRoleRevokedHandler?.();
      }
    } catch {}
  }

  return res;
}
