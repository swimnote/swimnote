============================================================

hyphen:
010-1234-5678

Step 2-A:

기존:
WHERE phone = kakaoPhone
AND swimming_pool_id = requestPoolId

수정:
WHERE (phone = kakaoPhone OR phone = kakaoPhoneHyphenP)
AND swimming_pool_id = requestPoolId

Step 2-B:

기존:
WHERE phone = kakaoPhone
LIMIT 2

수정:
WHERE (phone = kakaoPhone OR phone = kakaoPhoneHyphenP)
LIMIT 2

Step 4:

기존:
WHERE phone = kakaoPhone
AND role IN ('teacher', 'pool_admin')
LIMIT 1

수정:
WHERE (phone = kakaoPhone OR phone = kakaoPhoneHyphen)
AND role IN ('teacher', 'pool_admin')
LIMIT 1

중요:
기존 helper/변수명이 실제 Production 코드와 다르면
